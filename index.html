async function findStudentFile(prefix) { // Check cache first if (folderCache.has(prefix)) { return folderCache.get(prefix); } // Try folders from 1 to MAX_FOLDERS_TO_CHECK for (let i = 1; i <= MAX_FOLDERS_TO_CHECK; i++) { const folderName = `data_${i}`; const filePath = `data/${folderName}/${prefix}.json`; try { const response = await fetch(filePath, { method: 'HEAD' }); if (response.ok) { folderCache.set(prefix, filePath); return filePath; } } catch (err) { continue; } } // Fallback: try GET requests if HEAD fails for (let i = 1; i <= MAX_FOLDERS_TO_CHECK; i++) { const folderName = `data_${i}`; const filePath = `data/${folderName}/${prefix}.json`; try { const response = await fetch(filePath); if (response.ok) { folderCache.set(prefix, filePath); return filePath; } } catch (err) { continue; } } folderCache.set(prefix, null); return null; }
